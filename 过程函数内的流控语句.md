# 过程函数内的流控语句



## 选择结构

选择结构（分支结构）：程序按照条件进行选择执行。可以从两个或者多个分路中选择一个满足条件的路线执行



### IF

**格式：**（需要在存储过程或存储函数中写）

```sql
IF 表达式1 THEN 操作1
ELSEIF 表达式2 THEN 操作2
ELSEIF 表达式3 THEN 操作3
......
ELSE 操作N（上述的条件全部不匹配的时候，执行操作N）
END IF
```

>   在 MySQL 的 IF / ELSEIF 判断语句里，可以用三种东西：
>
>   *   自己 DECLARE 声明的变量
>
>   *   系统变量（比如 @@version 等）
>
>   *   直接写常量或表达式（比如 IF 1 = 1 THEN）
>
>   **但唯独不能直接写表里的列名（比如直接写 commission_pct）**







**示例：**（声明存储过程“update_salary_by_eid1”，定义IN参数emp_id，输入员工编号。判断该员工薪资如果低于8000元并且入职时间超过5年，就涨薪500元；否则就不变。）

```sql
DELIMITER $$
CREATE PROCEDURE update_salary_by_eid1(IN emp_id INT)
BEGIN
  DECLARE sal DOUBLE;
  DECLARE da DOUBLE;
  SELECT salary INTO sal FROM employees WHERE employee_id = emp_id;
  SELECT DATEDIFF(CURDATE(),hire_date)/365 INTO da FROM employees WHERE employee_id = emp_id;
  
  IF sal < 8000 AND da > 5 # 可以看到，必须是过程函数内的才能写在判断语句内
         THEN UPDATE employees SET salary = salary + 500 WHERE employee_id = emp_id; # 涨薪500
  END IF;
END$$
DELIMITER ;

CALL update_salary_by_eid1(101) # 想看可以去看下
SELECT salary FROM employees WHERE employee_id = 101;
```



**示例：**（声明存储过程“update_salary_by_eid3”，定义IN参数emp_id，输入员工编号。判断该员工薪资如果低于9000元，就更新薪资为9000元；薪资如果大于等于9000元且低于10000的，但是奖金比例为NULL的，就更新奖金比例为0.01；其他的涨薪100元。）

```sql
DELIMITER $$
CREATE PROCEDURE update_salary_by_eid6(IN emp_id INT)
BEGIN
    DECLARE sal DOUBLE;
    DECLARE da DOUBLE;
    DECLARE bili DOUBLE; # 只要出现在判断语句里面的变量都得声明
    SELECT salary INTO sal FROM employees WHERE employee_id = emp_id;
    SELECT DATEDIFF(CURDATE(),hire_date) INTO da FROM employees WHERE employee_id = emp_id;
    SELECT commission_pct INTO bili FROM employees WHERE employee_id = emp_id; # 给他们三个变量赋值
    
    IF sal < 9000
        THEN UPDATE employees SET salary = 9000 WHERE employee_id = emp_id;
    ELSEIF bili IS NULL && sal >= 9000 && sal <10000
        THEN UPDATE employees SET commission_pct = 0.01 WHERE employee_id = emp_id;
    ELSE UPDATE employees SET salary = salary + 100 WHERE employee_id = emp_id;
    END IF;
    
END $$
DELIMITER ;
    
SELECT salary,commission_pct FROM employees WHERE employee_id =  103;
CALL update_salary_by_eid6(103);
```





### CASE



**格式：**（过程函数之中）（当然有两种）

```
CASE 变量
WHEN 值1 THEN 结果1
WHEN 值2 THEN 结果2
ELSE 结果N
END CASE ;
```

```
CASE 
WHEN 判断语句1 THEN 语句1;
WHEN 判断语句2 THEN 语句2;
ELSE 语句N;
END CASE ;
```





**示例：**（声明存储过程“update_salary_by_eid4”，定义IN参数emp_id，输入员工编号。判断该员工薪资如果低于9000元，就更新薪资为9000元；薪资大于等于9000元且低于10000的，但是奖金比例为NULL的，就更新奖金比例为0.01；其他的涨薪100元。）

```sql
DELIMITER $$
CREATE PROCEDURE update_salary_by_eid7(IN emp_id INT)
BEGIN
   DECLARE emp_salary DOUBLE;
   DECLARE bili DOUBLE;
   SELECT salary INTO emp_salary FROM employees WHERE employee_id = emp_id;
   SELECT commission_pct INTO bili FROM employees WHERE employee_id = emp_id;

  CASE 
  WHEN emp_salary < 9000 THEN  
    UPDATE employees SET salary = 9000 WHERE employee_id = emp_id;
  WHEN emp_salary < 10000 AND bili IS NULL THEN
    UPDATE employees SET commission_pct = 0.1 WHERE employee_id = emp_id;
  ELSE
    UPDATE employees SET salary = salary + 100 WHERE employee_id = emp_id;
 END CASE;

END $$
DELIMITER ;

CALL update_salary_by_eid7(120);
SELECT salary FROM employees WHERE employee_id = 120;
```













## 循环结构

循环结构：程序满足一定的条件的时候，重复的连续的执行某些语句



循环LOOP\REPEAT\WHILE

### LOOP

LOOP：指的是重复执行LOOP循环体中的语句，直到被认为的通过关键字退出 LEAVE



**格式：**

```sql
loop_label: LOOP
	......
IF exit_condition THEN
	LEAVE loop_label; 
END IF;
END LOOP loop_label;
```





**示例：**

```sql
DELIMITER $$
CREATE PROCEDURE test9()
BEGIN
  DECLARE num1 INT DEFAULT 1;
  nailong:LOOP 
    SELECT num1;
    SET num1 = num1 + 1;
    IF num1 >= 11
    	THEN LEAVE loop_label;
    END IF;
END LOOP nailong;
```





**示例：**（“update_salary_loop()”，声明OUT参数num，输出循环次数。存储过程中实现循环给大家涨薪，薪资涨为

原来的1.1倍。直到全公司的平均薪资达到12000结束。并统计循环次数。）

```sql
DELIMITER $$
CREATE  PROCEDURE update_salary_loop(OUT num INT)
BEGIN
  DECLARE avg_salary DOUBLE;
  DECLARE loop_num INT DEFAULT 0;
  SELECT AVG(salary) INTO avg_salary FROM employees;
  loop_label:LOOP 
    IF avg_salary >= 12000 THEN LEAVE loop_label;
    END IF;
    UPDATE employees SET salary = salary * 1.1;
    SET loop_num = loop_num + 1;
    SELECT AVG(salary) INTO avg_salary FROM employees;
  END LOOP loop_label;
  SET num = loop_num;
  
END $$
DELIMITER ;

CALL update_salary_loop(@num);
SELECT @num;
```

-- 循环：

-- 1、初始化条件 2、循环成立的条件 3、循环体 4、迭代条件（初始化的条件进行更改）

--  市场环境不好时，公司为了渡过难关，决定暂时降低大家的薪资。声明存储过程

-- “update_salary_while()”，声明OUT参数num，输出循环次数。存储过程中实现循环给大家降薪，薪资降

-- 为原来的90%。直到全公司的平均薪资达到5000结束。并统计循环次数。

DROP PROCEDURE update_salary_while;

DELIMITER $$

CREATE PROCEDURE update_salary_while(OUT num INT)

BEGIN

  DECLARE avg_salary DOUBLE DEFAULT 0.0;

  DECLARE while_count INT DEFAULT 0 ;

  

  SELECT AVG(salary) INTO avg_salary FROM employees;

  

  while_label:WHILE  avg_salary > 5000 DO

  UPDATE employees SET salary = salary * 0.9;

  SELECT AVG(salary) INTO avg_salary FROM employees;

  SET while_count = while_count + 1;

  

  END WHILE while_label;

  

  SET num = while_count;

END $$

DELIMITER ;

CALL update_salary_while(@num);

SELECT @num;

-- 声明存储过程

-- “update_salary_repeat()”，声明OUT参数num，输出循环次数。存储过程中实现循环给大家涨薪，薪资涨

-- 为原来的1.15倍。直到全公司的平均薪资达到13000结束。并统计循环次数

DELIMITER $$

CREATE PROCEDURE update_salary_repeat(OUT num INT)

BEGIN

  DECLARE salary_avg DOUBLE;

  DECLARE while_count INT DEFAULT 0 ;

  

  SELECT AVG(salary) INTO salary_avg FROM employees;

  

  REPEAT

​    UPDATE employees SET salary = salary * 1.15;

​    SET while_count = while_count + 1;

​    SELECT AVG(salary) INTO salary_avg FROM employees;

​    

  UNTIL salary_avg>=13000

  

  END REPEAT;

  

  SET num = while_count;

END $$

DELIMITER ;

CALL update_salary_repeat(@num1);

SELECT @num1;
